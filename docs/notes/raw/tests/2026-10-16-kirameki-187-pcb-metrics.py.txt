import sys, numpy as np, subprocess, tempfile, os
from scipy import signal
from scipy.io import wavfile
R0=29762
def load(p):
    if p.endswith('.flac'):
        t=tempfile.mktemp(suffix='.wav'); subprocess.run(['ffmpeg','-y','-loglevel','error','-i',p,t],check=True); p=t
    r,s=wavfile.read(p); s=s.astype(float)/32768
    return signal.resample_poly(s,R0,r,axis=0) if r!=R0 else s
def corr(a,b): return float(a@b/np.sqrt((a@a)*(b@b)+1e-30))
def onset(x): return int(np.nonzero(np.abs(x).max(1)>0.003)[0][0])
def db(x): return 10*np.log10(x+1e-20)
def seg(x,o,t0,t1): return x[o+int(t0*R0):o+int(t1*R0)]
def metrics(x,o,lagfix=0):
    m={}
    a=seg(x,o,3,19.5); L,R=a[:,0],a[:,1]
    m['rms_L_dB']=db(L@L/len(L)); m['rms_R_dB']=db(R@R/len(R))
    m['LR_corr']=corr(L,R)
    M=L+R;S=L-R; m['S/M_dB']=db(S@S/(M@M))
    # lagged inter-channel xcorr beyond 5 ms (reverb-like cross-feed)
    l=L-L.mean(); r=R-R.mean()
    cc=signal.correlate(r,l,'full',method='fft')/np.sqrt((l@l)*(r@r)); lg=signal.correlation_lags(len(r),len(l))
    k=(lg>150)&(lg<3000); m['xcorr_L->R_5-100ms']=float(np.abs(cc[k]).max())
    k=(lg<-150)&(lg>-3000); m['xcorr_R->L_5-100ms']=float(np.abs(cc[k]).max())
    # tail after song end (end = last sample > -45 dBFS in 100ms envelope near 19.8s)
    pre=seg(x,o,17.5,19.7); pe=(pre**2).mean()
    w=int(.2*R0); tl=[db((seg(x,o,t,t+.2)**2).mean())-db(pe) for t in (19.9,20.1,20.3,20.5,20.9,21.3)]
    m['tail_rel_dB@19.9,20.1,20.3,20.5,20.9,21.3']=[round(v,1) for v in tl]
    # third-octave-ish band balance: energy 0-1k,1-4k,4-8k,8k+ relative to total
    f,P=signal.welch(a[:,0]+a[:,1],R0,nperseg=4096); tot=P.sum()
    m['band_dB(<1k,1-4k,4-8k,>8k)']=[round(float(db(P[(f>=lo)&(f<hi)].sum()/tot)),1) for lo,hi in ((0,1000),(1000,4000),(4000,8000),(8000,15000))]
    return m
def pcb_vs(pcb,x,op,ox):
    out=[]
    for t0 in range(3,19,3):
        n=3*R0; a=pcb[op+t0*R0:op+t0*R0+n]; M=a.sum(1)
        c0=ox+t0*R0-2500; b=x[c0:c0+n+5000]
        cc=signal.correlate(b.sum(1),M,'valid',method='fft'); k=int(np.argmax(cc)); bb=b[k:k+n]
        Sa=a[:,0]-a[:,1]; Sb=bb[:,0]-bb[:,1]
        out.append((corr(M,bb.sum(1)),corr(Sa,Sb)))
    return np.mean([o[0] for o in out]),np.mean([o[1] for o in out])

def feats(p):
    a=resp(p); E=(a**2).sum(1); n=len(E)
    tot=E.sum(); direct=E[:int(.01*R0)].sum(); 
    tail=E[int(.03*R0):].sum()
    c=np.cumsum(E[::-1])[::-1]
    def e(t): return 10*np.log10(c[min(int(t*R0),n-1)]/tot+1e-20)
    # decay rate of the instantaneous envelope between .3s and 1.5s (dB/s) via smoothed power
    w=int(.1*R0); pw=np.array([E[i:i+w].sum() for i in range(0,min(n,int(3*R0))-w,w)]); db=10*np.log10(pw+1e-20)
